# Next.js App Scaffold — Fantasy Hoops

Below is a minimal **Next.js 14 (App Router) + TypeScript + Tailwind** project that adapts the single‑file prototype into a proper app structure with:

- **App Router pages** (`app/page.tsx`, `app/player/[id]/page.tsx` stub)
- A demo **/api/players** endpoint returning mock data
- **Modularized code** (`lib/types`, `lib/mockPlayers`, `lib/scoring`)
- Lightweight **UI components** under `components/ui` so it runs without extra setup (you can replace these with shadcn later)

> Copy these files into a fresh `npx create-next-app@latest` project. If you want, I can export a zip.

### `lib/scoring.ts`
```ts
import { Player } from "./types";

export function fantasyScore(p: Player, weights: Record<string, number>) {
  const s = p.statsPerGame;
  return (
    s.pts * (weights.pts ?? 1) +
    s.reb * (weights.reb ?? 1) +
    s.ast * (weights.ast ?? 1) +
    s.stl * (weights.stl ?? 1.5) +
    s.blk * (weights.blk ?? 1.5) +
    s.tpm * (weights.tpm ?? 1.1) +
    (s.fg - 0.45) * 100 * (weights.fg ?? 0.5) +
    (s.ft - 0.75) * 100 * (weights.ft ?? 0.5) -
    s.tov * (weights.tov ?? 0.7)
  );
}
```

import React from "react";
export function Slider({ value, defaultValue, onValueChange, min = 0, max = 100, step = 1 }: any) {
  const [v, setV] = React.useState<number>(defaultValue?.[0] ?? value?.[0] ?? 0);
  return (
    <input
      type="range"
      min={min}
      max={max}
      step={step}
      value={v}
      onChange={(e) => { const nv = Number(e.target.value); setV(nv); onValueChange?.([nv]); }}
      className="w-full"
    />
  );
}

export default function Page() {
  const [players, setPlayers] = useState<Player[]>([]);
  const [query, setQuery] = useState("");
  const [team, setTeam] = useState<string>("ALL");
  const [pos, setPos] = useState<string>("ALL");
  const [scoring, setScoring] = useState<string>("9-cat");
  const [sortKey, setSortKey] = useState<keyof Player | "score" | "adp">("score");
  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");
  const [openPlayer, setOpenPlayer] = useState<Player | null>(null);
  const [weights, setWeights] = useState<Record<string, number>>({});

  useEffect(() => {
    fetch("/api/players").then(r => r.json()).then(({ players }) => setPlayers(players));
  }, []);

  const teams: Team[] = [
    "ATL","BOS","BKN","CHA","CHI","CLE","DAL","DEN","DET","GSW",
    "HOU","IND","LAC","LAL","MEM","MIA","MIL","MIN","NOP","NYK",
    "OKC","ORL","PHI","PHX","POR","SAC","SAS","TOR","UTA","WAS",
  ];

  const filtered = useMemo(() => {
    const base = players.map((p) => ({ ...p, score: fantasyScore(p, weights) }));
    return base
      .filter((p) => (team === "ALL" ? true : p.team === team))
      .filter((p) => (pos === "ALL" ? true : p.pos.includes(pos as Position)))
      .filter((p) => p.name.toLowerCase().includes(query.toLowerCase()));
  }, [players, team, pos, query, weights]);

  const sorted = useMemo(() => {
    const arr: any[] = [...filtered];
    arr.sort((a, b) => {
      const A = sortKey === "score" ? a.score : sortKey === "adp" ? a.adp : (a as any)[sortKey];
      const B = sortKey === "score" ? b.score : sortKey === "adp" ? b.adp : (b as any)[sortKey];
      if (A < B) return sortDir === "asc" ? -1 : 1;
      if (A > B) return sortDir === "asc" ? 1 : -1;
      return 0;
    });
    return arr;
  }, [filtered, sortKey, sortDir]);

  function header(h: string, key: keyof Player | "score" | "adp") {
    const active = sortKey === key;
    const DirIcon = active && sortDir === "asc" ? ChevronUp : ChevronDown;
    return (
      <th className="px-3 py-2 text-left text-sm font-semibold text-zinc-700">
        <button
          onClick={() => {
            if (sortKey === key) setSortDir((d) => (d === "asc" ? "desc" : "asc"));
            else { setSortKey(key); setSortDir("desc"); }
          }}
          className="inline-flex items-center gap-1 hover:opacity-80"
        >
          {h}
          <DirIcon className={`h-4 w-4 ${active ? "opacity-100" : "opacity-30"}`} />
        </button>
      </th>
    );
  }

  return (
    <TooltipProvider>
      <div className="min-h-screen w-full">
        {/* Top Nav */}
        <div className="sticky top-0 z-30 border-b bg-white/80 backdrop-blur">
          <div className="mx-auto max-w-7xl px-4 py-3 flex items-center justify-between">
            <div className="flex items-center gap-3">
              <BarChart2 className="h-6 w-6" />
              <span className="text-xl font-bold tracking-tight">Fantasy Hoops</span>
              <span className="text-xs rounded-full bg-zinc-100 px-2 py-0.5 ml-2">alpha</span>
            </div>
            <div className="flex items-center gap-2">
              <Button variant="ghost" size="icon"><HelpCircle className="h-5 w-5" /></Button>
              <Button variant="ghost" size="icon"><Settings2 className="h-5 w-5" /></Button>
            </div>
          </div>
        </div>

        <main className="mx-auto max-w-7xl px-4 py-6">
          <Tabs defaultValue="rankings">
            <TabsList className="grid w-full max-w-md grid-cols-3">
              <TabsTrigger value="rankings" className="flex gap-2"><Star className="h-4 w-4" />Rankings</TabsTrigger>
              <TabsTrigger value="teams" className="flex gap-2"><Users className="h-4 w-4" />Teams</TabsTrigger>
              <TabsTrigger value="tools" className="flex gap-2"><BarChart2 className="h-4 w-4" />Tools</TabsTrigger>
            </TabsList>

            {/* Rankings Tab */}
            <TabsContent value="rankings" className="mt-6">
              <Card className="shadow-sm">
                <CardHeader className="pb-2">
                  <CardTitle className="text-lg">Player Rankings & Projections</CardTitle>
                </CardHeader>
                <CardContent className="space-y-4">
                  {/* Controls */}
                  <div className="grid grid-cols-1 gap-3 md:grid-cols-6">
                    <div className="md:col-span-2 flex items-center gap-2">
                      <Search className="h-4 w-4 text-zinc-500" />
                      <Input value={query} onChange={(e) => setQuery(e.target.value)} placeholder="Search players..." />
                    </div>
                    <div>
                      <Label className="text-xs">Team</Label>
                      <Select value={team} onValueChange={setTeam}>
                        <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
                        <SelectContent>
                          <SelectItem value="ALL">All Teams</SelectItem>
                          {teams.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}

                        <tr>
                          <th className="px-3 py-2 text-left text-sm font-semibold text-zinc-700">#</th>
                          {header("Player", "score")}
                          {header("Team", "team")}
                          <th className="px-3 py-2 text-left text-sm font-semibold text-zinc-700">POS</th>
                          {header("Score", "score")}
                          {header("ADP", "adp")}

                <CardContent className="space-y-3 text-sm text-zinc-700">
                  <ol className="list-decimal pl-5 space-y-2">
                    <li>Point your fetch to a real endpoint (e.g., Supabase/Firehose/your API).</li>
                    <li>Replace mock stats with model projections (9‑cat z‑scores, points weights).</li>
                    <li>Add ISR or caching for speed on Vercel.</li>
                    <li>Create `/player/[id]` for deep player pages with SEO.</li>
                  </ol>
                </CardContent>
              </Card>
            </TabsContent>
          </Tabs>
        </main>

        {/* Player Dialog */}
        <Dialog open={!!openPlayer} onOpenChange={() => setOpenPlayer(null)}>
          <DialogContent className="sm:max-w-2xl">
            <DialogHeader>
              <DialogTitle className="flex items-center justify-between">
                <span>{openPlayer?.name}</span>
                <Button size="icon" variant="ghost" onClick={() => setOpenPlayer(null)}><X className="h-5 w-5" /></Button>
              </DialogTitle>
            </DialogHeader>
            {openPlayer && (
              <div className="grid grid-cols-1 gap-4 md:grid-cols-5">
                <div className="md:col-span-2 space-y-2">
                  <InfoRow label="Team" value={openPlayer.team} />
                  <InfoRow label="Positions" value={openPlayer.pos.join("/")} />
                  <InfoRow label="ADP" value={openPlayer.adp.toFixed(1)} />
                  <InfoRow label="Minutes" value={`${openPlayer.minutes} mpg`} />
                  <div className="text-xs text-zinc-500">Fantasy Score (current weights): {fantasyScore(openPlayer, weights).toFixed(1)}</div>
                </div>
                <div className="md:col-span-3 h-56">
                  <ResponsiveContainer width="100%" height="100%">
                    <LineChart data={openPlayer.recentGameLog} margin={{ top: 8, right: 16, left: 0, bottom: 0 }}>
                      <XAxis dataKey="g" tick={{ fontSize: 12 }} />
                      <YAxis tick={{ fontSize: 12 }} />
                      <RTooltip />
                      <Legend />
                      <Line type="monotone" dataKey="pts" strokeWidth={2} dot={false} />
                      <Line type="monotone" dataKey="reb" strokeWidth={2} dot={false} />
                      <Line type="monotone" dataKey="ast" strokeWidth={2} dot={false} />
                    </LineChart>
                  </ResponsiveContainer>
                </div>
              </div>
            )}
          </DialogContent>
        </Dialog>

        <footer className="mx-auto max-w-7xl px-4 py-12 text-xs text-zinc-500">Built with ❤️ in Next.js & Tailwind.</footer>
      </div>
    </TooltipProvider>
  );
}

function InfoRow({ label, value }: { label: string; value: React.ReactNode }) {
  return (
    <div className="flex items-center justify-between rounded-lg bg-zinc-50 px-3 py-2">
      <span className="text-xs text-zinc-500">{label}</span>
      <span className="text-sm font-medium">{value}</span>
    </div>
  );
}

function WeightRow({ label, k, weights, setWeights }: { label: string; k: string; weights: Record<string, number>; setWeights: (u: any) => void }) {
  const v = (weights[k] ?? 1) * 100;
  return (
    <div className="space-y-1">
      <div className="flex items-center justify-between">
        <Label>{label}</Label>
        <Tooltip>
          <TooltipTrigger asChild>
            <span className="text-xs text-zinc-500">{(v / 100).toFixed(2)}x</span>
          </TooltipTrigger>
          <TooltipContent>Multiply contribution by this factor.</TooltipContent>
        </Tooltip>
      </div>
      <Slider value={[v]} max={300} min={0} step={5} onValueChange={([nv]) => setWeights((w: any) => ({ ...w, [k]: nv / 100 }))} />
    </div>
  );
}

## Quick Start
1. `npx create-next-app fantasy-hoops --ts`

3. `npm i lucide-react recharts tailwindcss postcss autoprefixer`
4. `npm run dev` → open http://localhost:3000

When you’re ready, we can:
- Swap the lightweight UI for **shadcn/ui** and move styles to Radix-based components.
- Hook `/api/players` to a real data source.
- Add `/draft`, `/schedule`, and team/player SEO pages with metadata.
